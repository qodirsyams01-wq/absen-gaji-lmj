// GANTI TAB
function bukaTab(nama) {
    document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('aktif'));
    event.target.closest('.tab-btn')?.classList.add('aktif');
    ['scan','karyawan','absen','laporan','setelan'].forEach(t => {
        document.getElementById(`tab-${t}`).classList.add('hidden');
    });
    document.getElementById(`tab-${nama}`).classList.remove('hidden');
    
    if (nama === 'karyawan') tampilDaftarTukang();
    if (nama === 'absen') tampilAbsen();
    if (nama === 'laporan') laporanMode('harian');
}

// SIMPAN URL WEB APP
function simpanUrl() {
    urlWebApp = document.getElementById('urlWebApp').value.trim();
    if (!urlWebApp) {
        alert('⚠️ Tempel URL Web App dari Google Apps Script!');
        return;
    }
    localStorage.setItem('lmj_url_app', urlWebApp);
    document.getElementById('statusSinkron').textContent = '✅ Terhubung ke Google Sheets';
    document.getElementById('statusSinkron').className = 'text-xs mt-2 text-green-400';
    document.getElementById('pesanKoneksi').innerHTML = '<span class="sukses">✅ URL tersimpan! Data akan langsung dikirim ke Google Sheets saat absen.</span>';
}

// KIRIM DATA KE GOOGLE SHEETS
async function kirimKeSheet(data) {
    if (!urlWebApp) {
        alert('⚠️ Masukkan URL Web App dulu di tab Setelan!');
        return false;
    }
    try {
        const res = await fetch(urlWebApp, {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify(data)
        });
        const hasil = await res.json();
        return hasil.hasil === 'berhasil';
    } catch (e) {
        console.error(e);
        return false;
    }
}

// TAMBAH TUKANG
function simpanTukang() {
    const nama = document.getElementById('namaBaru').value.trim();
    const nik = document.getElementById('nikBaru').value.trim();
    const jabatan = document.getElementById('jabatanBaru').value;
    const gaji = parseInt(document.getElementById('gajiBaru').value) || 0;

    if (!nama || !nik || gaji === 0) {
        alert('⚠️ Semua kolom wajib diisi!');
        return;
    }
    if (daftarTukang.find(t => t.nik === nik)) {
        alert('⚠️ NIK sudah terdaftar!');
        return;
    }

    const kodeBarcode = `LMJ-${nik}-${Date.now().toString().slice(-6)}`;
    const baru = {
        id: Date.now(), nama, nik, jabatan, gajiPerHari: gaji,
        kodeBarcode, tanggalDaftar: new Date().toLocaleDateString('id-ID'),
        totalHadir: 0, totalPotongan: 0
    };

    daftarTukang.push(baru);
    localStorage.setItem('lmj_tukang', JSON.stringify(daftarTukang));
    
    // Kirim ke Sheet
    kirimKeSheet({
        aksi: 'simpan_karyawan',
        id: baru.id, nama, nik, jabatan, kodeBarcode, gajiPerHari: gaji,
        tanggalDaftar: baru.tanggalDaftar
    });

    // Tampilkan Barcode
    document.getElementById('barcodeTukang').classList.remove('hidden');
    document.getElementById('kodeTukang').textContent = kodeBarcode;
    setTimeout(() => {
        JsBarcode('#bc-hasil', kodeBarcode, {format:"CODE128", width:2, height:60, displayValue:true});
    }, 100);

    alert(`✅ Tersimpan!\nNama: ${nama}\nGaji/Hari: Rp ${gaji.toLocaleString('id-ID')}`);
    
    document.getElementById('namaBaru').value = '';
    document.getElementById('nikBaru').value = '';
    document.getElementById('gajiBaru').value = '';
    tampilDaftarTukang();
}

function tampilDaftarTukang() {
    const el = document.getElementById('daftarTukang');
    if (daftarTukang.length === 0) {
        el.innerHTML = '<p class="text-gray-400 text-center py-4">Belum ada data tukang</p>';
        return;
    }
    el.innerHTML = daftarTukang.map(t => `
        <div class="info-kartu flex justify-between items-center">
            <div>
                <b>${t.nama}</b> — ${t.jabatan}<br>
                <span class="text-xs text-gray-400">Rp ${t.gajiPerHari.toLocaleString('id-ID')}/hari • Hadir: ${t.totalHadir} hari</span>
            </div>
            <div class="text-right">
                <button onclick="hapusTukang(${t.id})" class="text-red-400 text-xs">🗑️</button>
            </div>
        </div>
    `).join('');
}

function hapusTukang(id) {
    if (!confirm('Hapus data tukang ini?')) return;
    daftarTukang = daftarTukang.filter(t => t.id !== id);
    localStorage.setItem('lmj_tukang', JSON.stringify(daftarTukang));
    tampilDaftarTukang();
}

// SCAN BARCODE
async function bukaKameraScan() {
    document.getElementById('modalScan').classList.add('modal-buka');
    kameraAktif = true;
    
    try {
        const stream = await navigator.mediaDevices.getUserMedia({ 
            video: { facingMode: 'environment', width: { ideal: 1280 } } 
        });
        const video = document.getElementById('videoScan');
        video.srcObject = stream;
        await video.play();

        Quagga.init({
            inputStream: { type: 'LiveStream', target: video, constraints: { width: 640, height: 480 } },
            decoder: { readers: ['code_128_reader'] }
        }, err => {
            if (err) {
                const kode = prompt('Masukkan kode barcode secara manual:');
                if (kode) prosesHasilScan(kode);
                return;
            }
            Quagga.start();
            Quagga.onDetected(data => {
                Quagga.stop();
                tutupScan();
                prosesHasilScan(data.codeResult.code);
            });
        });
    } catch (err) {
        const kode = prompt('Tidak bisa akses kamera.\nMasukkan kode barcode:');
        if (kode) prosesHasilScan(kode);
    }
}

function tutupScan() {
    kameraAktif = false;
    const video = document.getElementById('videoScan');
    if (video.srcObject) video.srcObject.getTracks().forEach(t => t.stop());
    try { Quagga.stop(); } catch(e) {}
    document.getElementById('modalScan').classList.remove('modal-buka');
}

async function prosesHasilScan(kodeBarcode) {
    const tukang = daftarTukang.find(t => t.kodeBarcode === kodeBarcode);
    if (!tukang) {
        alert('❌ Barcode tidak terdaftar!');
        return;
    }

    const sekarang = new Date();
    const tanggal = sekarang.toISOString().split('T')[0];
    const jam = sekarang.toLocaleTimeString('id-ID', {hour:'2-digit', minute:'2-digit'});
    const jamMasuk = sekarang.getHours();
    const status = jamMasuk > 8 ? 'Terlambat' : 'Hadir';
    const potongan = status === 'Terlambat' ? 5000 : 0;

    // Cek sudah absen hari ini?
    const sudah = dataAbsen.find(a => a.tanggal === tanggal && a.kodeBarcode === kodeBarcode);
    if (sudah) {
        alert(`⚠️ ${tukang.nama} sudah absen hari ini!\nJam Masuk: ${sudah.jamMasuk}`);
        return;
    }

    // Simpan absen
    const absenBaru = {
        id: Date.now(),
        kodeBarcode,
        idTukang: tukang.id,
        nama: tukang.nama,
        tanggal,
        jamMasuk: jam,
        status,
        potongan,
        gajiPerHari: tukang.gajiPerHari,
        proyek: document.getElementById('pilihProyekScan').value,
        waktu: sekarang.toLocaleString('id-ID')
    };

    dataAbsen.unshift(absenBaru);
    localStorage.setItem('lmj_absen', JSON.stringify(dataAbsen));

    // Update total hadir tukang
    tukang.totalHadir++;
    tukang.totalPotongan += potongan;
    localStorage.setItem('lmj_tukang', JSON.stringify(daftarTukang));

    // Kirim ke Google Sheets
    await kirimKeSheet({
        aksi: 'simpan_absen',
        timestamp: absenBaru.waktu,
        idKaryawan: tukang.id,
        nama: tukang.nama,
        tanggal,
        jamMasuk: jam,
        status,
        potongan,
        idProyek: absenBaru.proyek,
        namaProyek: document.getElementById('pilihProyekScan').selectedOptions[0].text
    });

    // Tampilkan hasil
    document.getElementById('hasilScan').classList.remove('hidden');
    document.getElementById('hnama').textContent = tukang.nama;
    document.getElementById('hjam').textContent = jam;
    document.getElementById('hstatus').textContent = status + (potongan ? ` (Potong Rp ${potongan.toLocaleString('id-ID')})` : '');
    document.getElementById('hgaji').textContent = tukang.gajiPerHari.toLocaleString('id-ID');

    alert(`✅ Absen Tercatat!\nNama: ${tukang.nama}\nJam: ${jam}\nStatus: ${status}`);
    tampilAbsen();
}

// TAMPIL ABSEN
function tampilAbsen() {
    const tgl = document.getElementById('filterTanggal').value;
    const hariIni = dataAbsen.filter(a => a.tanggal === tgl);
    const el = document.getElementById('isiAbsen');
    
    if (hariIni.length === 0) {
        el.innerHTML = '<p class="text-gray-400 text-center py-4">Belum ada absen pada tanggal ini</p>';
        return;
    }

    el.innerHTML = hariIni.map(a => `
        <div class="info-kartu flex justify-between items-center">
            <div>
                <b>${a.nama}</b><br>
                <span class="text-xs ${a.status === 'Hadir' ? 'text-green-400' : 'text-yellow-400'}">
                    ${a.status} • ${a.jamMasuk}
                </span>
            </div>
            <div class="text-right">
                <span class="text-sm">Rp ${(a.gajiPerHari - a.potongan).toLocaleString('id-ID')}</span><br>
                <span class="text-xs text-gray-400">Potong: Rp ${a.potongan.toLocaleString('id-ID')}</span>
            </div>
        </div>
    `).join('');
}

// LAPORAN GAJI
let modeLaporan = 'harian';
function laporanMode(mode) {
    modeLaporan = mode;
    ['harian','mingguan','bulanan'].forEach(m => {
        document.getElementById(`btn-${m}`).classList.remove('aktif');
    });
    document.getElementById(`btn-${mode}`).classList.add('aktif');
    hitungLaporan();
}

function hitungLaporan() {
    let filter = dataAbsen;
    let judul = '';

    if (modeLaporan === 'harian') {
        const tgl = new Date().toISOString().split('T')[0];
        filter = dataAbsen.filter(a => a.tanggal === tgl);
        judul = 'Hari Ini (' + new Date().toLocaleDateString('id-ID') + ')';
    } else if (modeLaporan === 'mingguan') {
        const hari = new Date().getDay();
        const mulai = new Date(); mulai.setDate(mulai.getDate() - hari);
        const akhir = new Date(); akhir.setDate(akhir.getDate() + (6 - hari));
        filter = dataAbsen.filter(a => {
            const d = new Date(a.tanggal);
            return d >= mulai && d <= akhir;
        });
        judul = 'Minggu Ini — Gajian Hari Sabtu';
    } else {
        const bln = new Date().getMonth();
        const thn = new Date().getFullYear();
        filter = dataAbsen.filter(a => {
            const d = new Date(a.tanggal);
            return d.getMonth() === bln && d.getFullYear() === thn;
        });
        judul = new Date().toLocaleString('id-ID', {month:'long', year:'numeric'});
    }

    // Rekap per tukang
    const rekap = {};
    filter.forEach(a => {
        if (!rekap[a.kodeBarcode]) {
            rekap[a.kodeBarcode] = {nama:a.nama, hadir:0, potongan:0, gajiPerHari:a.gajiPerHari};
        }
        if (a.status !== 'Tidak Hadir') rekap[a.kodeBarcode].hadir++;
        rekap[a.kodeBarcode].potongan += a.potongan;
    });

    let totalSemua = 0;
    const isi = Object.values(rekap).map(r => {
        const total = r.hadir * r.gajiPerHari - r.potongan;
        totalSemua += total;
        return `
        <div class="flex justify-between py-2 border-b border-white/10">
            <span><b>${r.nama}</b> — ${r.hadir} hari hadir</span>
            <span class="font-bold">Rp ${total.toLocaleString('id-ID')}</span>
        </div>`;
    }).join('');

    document.getElementById('isiLaporan').innerHTML = `
        <h3 class="font-bold text-lg mb-3">📊 ${judul}</h3>
        ${isi || '<p class="text-gray-400">Belum ada data</p>'}
        <div class="mt-4 pt-3 border-t border-white/20 text-xl font-bold text-green-400">
            💰 TOTAL: Rp ${totalSemua.toLocaleString('id-ID')}
        </div>
    `;
}

// UNDUH CSV
function simpanCSV() {
    let teks = "Nama,Tanggal,Jam Masuk,Status,Potongan,Gaji Bersih\n";
    dataAbsen.forEach(a => {
        teks += `"${a.nama}","${a.tanggal}","${a.jamMasuk}","${a.status}",${a.potongan},${a.gajiPerHari - a.potongan}\n`;
    });
    const blob = new Blob([teks], {type:'text/csv;charset=utf-8'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `Absen_Gaji_${new Date().toISOString().slice(0,10)}.csv`;
    a.click();
}

async function kirimKeSheet() {
    if (!urlWebApp) {
        alert('⚠️ Masukkan URL Web App dulu di tab Setelan!');
        return;
    }
    let berhasil = 0;
    for (const a of dataAbsen.slice(-50)) { // Kirim 50 data terakhir
        await kirimKeSheet({
            aksi: 'simpan_absen',
            timestamp: a.waktu,
            idKaryawan: a.idTukang,
            nama: a.nama,
            tanggal: a.tanggal,
            jamMasuk: a.jamMasuk,
            status: a.status,
            potongan: a.potongan,
            idProyek: a.proyek,
            namaProyek: ''
        });
        berhasil++;
    }
    alert(`✅ ${berhasil} data dikirim ke Google Sheets!`);
}
